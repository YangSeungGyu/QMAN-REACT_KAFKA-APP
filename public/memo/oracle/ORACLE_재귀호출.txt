================[샘플데이터]================
WITH DEPT AS (
    SELECT '10' AS DEPTNO, NULL AS UPPER_DEPTNO, '기획팀' AS DNAME, '서울' AS LOC FROM DUAL UNION ALL
    SELECT '20' AS DEPTNO, '10' AS UPPER_DEPTNO, '개발팀' AS DNAME, '판교' AS LOC FROM DUAL UNION ALL
    SELECT '30' AS DEPTNO, '10' AS UPPER_DEPTNO, '영업팀' AS DNAME, '부산' AS LOC FROM DUAL UNION ALL
    SELECT '40' AS DEPTNO, '20' AS UPPER_DEPTNO, '디자인팀' AS DNAME, '서울' AS LOC FROM DUAL UNION ALL
    SELECT '50' AS DEPTNO, '30' AS UPPER_DEPTNO, '마케팅팀' AS DNAME, '대전' AS LOC FROM DUAL UNION ALL
    SELECT '60' AS DEPTNO, '50' AS UPPER_DEPTNO, '마케팅 하위' AS DNAME, '동네' AS LOC FROM DUAL
)


================[활용1 [메뉴 레벨 확인]================
SELECT 
    --LTRIM(SYS_CONNECT_BY_PATH(DNAME, ' > '), ' > ') AS DEPT_PATH
    SYS_CONNECT_BY_PATH(DNAME, ' -> ') AS DEPT_PATH
FROM DEPT
START WITH UPPER_DEPTNO IS NULL
CONNECT BY PRIOR DEPTNO = UPPER_DEPTNO;

================[활용2 [메뉴 구조 생성]================
SELECT     
    LEVEL
    ,LPAD('A', (LEVEL - 1), 'A')
    ,LPAD(' ', (LEVEL - 1), ' ') || DNAME AS DEPT_MENU
FROM DEPT
START WITH UPPER_DEPTNO IS NULL
CONNECT BY PRIOR DEPTNO = UPPER_DEPTNO;
