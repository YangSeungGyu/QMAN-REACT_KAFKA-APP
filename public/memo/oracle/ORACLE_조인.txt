/*LEFT OUTER JOIN LATERAL 그 외 동일결과 사용법을 _2, _3으로 표기 
SELECT 
  DT.UPPER_DEPTNO

  ,EP2.LAST_HIREDATE AS LAST_HIREDATE_1

  ,(SELECT SUM(EP.SAL) AS TOT_SAL FROM EMP EP WHERE EP.DEPTNO = DT.DEPTNO) AS TOT_SAL_2
  ,(SELECT MAX(EP.HIREDATE) AS LAST_HIREDATE FROM EMP EP WHERE EP.DEPTNO = DT.DEPTNO) AS LAST_HIREDATE_2

================[CROSS JOIN 응용]================
/*다음 ROW값 참조 예제1차(중간 결과 참고)
SELECT 
  GT.TMP_NO
  ,(MT.RN +GT.TMP_NO) AS GROUP_KEY
  , MT.*
FROM
(SELECT LEVEL-1 AS TMP_NO FROM DUAL CONNECT BY LEVEL <= 2) GT
CROSS JOIN
(SELECT ROWNUM AS RN,EP.* FROM DEPT DT INNER JOIN  EMP EP ON DT.DEPTNO = EP.DEPTNO) MT
ORDER BY MT.RN
*/

/*다음 ROW값 참조 예제2차
SELECT AAA.* FROM
(
SELECT 
  MAX(MT.RN) AS RN
  ,MAX(DECODE(GT.TMP_NO,0,MT.SAL,0)) AS THIS_SAL
  ,MAX(DECODE(GT.TMP_NO,1,MT.SAL,0)) AS NEXT_SAL
  ,MAX(DECODE(GT.TMP_NO,0,EMPNO,NULL)) AS EMPNO
  ,MAX(DECODE(GT.TMP_NO,0,ENAME,NULL)) AS ENAME
FROM
(SELECT 0 AS TMP_NO FROM DUAL UNION SELECT 1 AS TMP_NO FROM DUAL) GT -- CONNECT BY LEVEL은 오라클용이고 일반적으로는 UNION 사용
CROSS JOIN
(SELECT ROWNUM AS RN,EP.* FROM DEPT DT INNER JOIN  EMP EP ON DT.DEPTNO = EP.DEPTNO) MT
GROUP BY (MT.RN +GT.TMP_NO)
) AAA
WHERE
AAA.EMPNO IS NOT NULL -- 매칭안되는 마지막 줄 삭제
ORDER BY AAA.RN
*/

/*동일 결과 LAG()OVER 오라클 함수 사용 - 단점 컬럼 하나씩만 됨. 세번째 인자는 NVL처리
SELECT 
    ROWNUM AS RN,
    SAL AS THIS_SAL,
    LAG(SAL,1, 0) OVER (ORDER BY ROWNUM) AS NEXT_SAL, -- 이전줄 참조는 LEAD() OVER()
    EMPNO,
    ENAME
FROM DEPT DT 
INNER JOIN EMP EP ON DT.DEPTNO = EP.DEPTNO
ORDER BY RN;
*/
