WITH DEPT AS (
    SELECT '10' AS DEPTNO, '기획팀' AS DNAME FROM DUAL UNION ALL
    SELECT '20' AS DEPTNO, '개발팀' AS DNAME FROM DUAL UNION ALL
    SELECT '30' AS DEPTNO, '영업팀' AS DNAME FROM DUAL
  
),
EMP AS (
  SELECT '김철수' AS ENAME, '10' AS DEPTNO, 5000 AS SAL FROM DUAL UNION ALL
  SELECT '조현우' AS ENAME, '20' AS DEPTNO, 5200 AS SAL FROM DUAL UNION ALL
  SELECT '이영희' AS ENAME, '30' AS DEPTNO, 6000 AS SAL FROM DUAL UNION ALL
  SELECT '박지성' AS ENAME, '10' AS DEPTNO, 4500 AS SAL FROM DUAL UNION ALL
  SELECT '이강인' AS ENAME, '20' AS DEPTNO, 7000 AS SAL FROM DUAL UNION ALL
  SELECT '최바다' AS ENAME, '30' AS DEPTNO, 3500 AS SAL FROM DUAL UNION ALL
  SELECT '정운전' AS ENAME, '20' AS DEPTNO, 4000 AS SAL FROM DUAL UNION ALL
  SELECT '손흥민' AS ENAME, '10' AS DEPTNO, 8000 AS SAL FROM DUAL UNION ALL
  SELECT '길동1' AS ENAME, '30' AS DEPTNO, 9000 AS SAL FROM DUAL UNION ALL
  SELECT '길동2' AS ENAME, '10' AS DEPTNO, 6000 AS SAL FROM DUAL UNION ALL

/*기본 합계
SELECT 
MAX(DT.DNAME) AS DNAME
,DT.DEPTNO
,SUM(EP.SAL) AS TOT_SAL
FROM DEPT DT
INNER JOIN EMP EP
ON
DT.DEPTNO = EP.DEPTNO
GROUP BY DT.DEPTNO
-- HAIVNG COUNT(*) > 1 추가 HAIVNG절

*/
