================[arrow function, 변수 function 차이]================
const obj = {
  name: "test",
  handleClick: function() {
    console.log(this.name); //this는 obj
  }
};

const obj = {
  name: "test",
  handleClick: () => {
    console.log(this.name); //this는 obj가 아님
  }
};


또한 함수 내용이 1줄일 소괄호()를 치면 return과 동일
const test = () => {return '11'}
const test = () => ('11')

================[arrow]================
setArrayObj(prev => [...prev, newObj]) (새 데이터 추가)
setArrayObj(prev => prev.map(...)) (특정 데이터 수정)
setArrayObj(prev => []) (전체 삭제)


// id가 'targetId'인 요소만 제외하고 나머지만 유지
setArrayObj(prev => prev.filter(item => item.id !== 'targetId'));


setArrayObj(prev => [newObj, ...prev]); //반대로 쓸 수 있으며 최상단으로 올림.


setArrayObj(prev => [newObj, ...prev].slice(0, 10));// 배열을 10개로 제한하고 신규객체를 추가

setArrayObj(prev => [...prev].sort((a, b) => a.price - b.price)); //오름차순 정렬
